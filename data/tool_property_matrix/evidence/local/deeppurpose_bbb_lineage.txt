def parse_request(value):
    if not isinstance(value, dict) or set(value) != {"invocation", "rows"}:
        raise ValueError("Worker request must contain only invocation and rows.")
    invocation = value["invocation"]
    if not isinstance(invocation, dict) or set(invocation) != {"round", "nonce"}:
        raise ValueError("Worker invocation challenge is malformed.")
    if invocation["round"] not in {"qualification", "discovery", "replay1", "replay2"}:
        raise ValueError
